# Plot the number of counties rated 'Relatively High' in each state with a line graph
lake_data |>
  arrange(desc(K)) |>
  slice_head(n = 10) |>
  ggplot(aes(x = reorder(LAKENAME, K),
             y = K)) +
  geom_linerange(aes(ymin = 0, ymax = K)) +
  geom_point(color = "red") +
  geom_text(aes(label = K, hjust = -0.8)) + 
  labs(x = "Lakes",
       y = "K concentration",
       title = "Lakes with the highest K concentations") +
  coord_flip() +
  theme_minimal() +
  theme(axis.title.y = element_blank())

# Top 10 lakes by NO3 concentration as a bar chart
